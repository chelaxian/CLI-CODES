# agents.txt -- реестр CLI-харнессов (AI-агентов), которые умеет ставить CLI-CODES.
#
# Формат (разделитель |, пустое поле = "-"):
#   key | name | command | linux-install | windows-install | adapter | legacy-launcher | homepage | description
#
# key              -- канонический ключ (используется в cli-codes <key>, ярлыках, state)
# command          -- имя исполняемого файла в PATH (проверка "установлен ли")
# linux-install /
# windows-install  -- способ установки:
#                       npm:<package>     npm install -g <package>@latest
#                       uv:<package>      uv tool install <package>   (uv ставится автоматически)
#                       sh:<url>          curl -fsSL <url> | bash      (только Linux/macOS)
#                       ps:<url>          irm <url> | iex              (только Windows)
#                       -                 не поддерживается на этой ОС
# adapter          -- как лаунчер подключает выбранного провайдера/модель:
#                       claude      ANTHROPIC_BASE_URL (нативно или через scripts/lib/anthropic-proxy.py)
#                       openclaude  CLAUDE_CODE_USE_OPENAI + OPENAI_*
#                       qwen        --auth-type openai --openai-base-url -m
#                       opencode    сгенерированный opencode.json (OPENCODE_CONFIG)
#                       codex       -c model_providers.* (только провайдеры с Responses API)
#                       gemini      GEMINI_API_KEY + -m (только Google AI Studio)
#                       aider       --openai-api-base --model openai/<id>
#                       goose       GOOSE_PROVIDER=openai + OPENAI_HOST/OPENAI_BASE_PATH
#                       env         провайдер не выбирается: агент использует свой логин/конфиг,
#                                   лаунчер лишь экспортирует сохранённые API-ключи в окружение
# legacy-launcher  -- старый специализированный лаунчер (scripts/<name>.sh|.ps1), "-" если нет
#
# Порядок строк = порядок в меню установщика и лаунчера.
